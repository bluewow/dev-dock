// .claude 관련 신규 메서드
class ProjectService {
  // 기존 메서드 유지...

  // Devdock 자체 .claude 경로 반환
  String get sourceClaudePath;

  // 대상 프로젝트에 .claude 존재 여부
  bool hasClaudeConfig(String projectPath);

  // .claude 설치 (settings.local.json 제외)
  Future<ClaudeInstallResult> installClaude({
    required String targetPath,
    bool forceOverwrite = false,
  });

  // .claude 파일 트리 스캔
  List<ClaudeFileEntry> scanClaudeFiles(
    String projectPath,
  );

  // .claude 파일 내용 읽기
  String readClaudeFile(String absolutePath);
}

// 설치 결과
class ClaudeInstallResult {
  final bool success;
  final int filesCopied;
  final String? error;
  final List<String> excludedFiles;
  // settings.local.json 가이드
  final String localSettingsGuide;
}

// 파일 트리 엔트리
class ClaudeFileEntry {
  final String name;
  final String relativePath;
  final String absolutePath;
  final bool isDirectory;
  final List<ClaudeFileEntry> children;
}

5 QA 체크 포인트 (놓친 점 검토)

필수
재귀 설치 방지
Devdock 프로젝트 자체를 등록할 때 .claude를 자기 자신에게 설치하면 안 됨. 경로 비교로 차단.
필수
쓰기 권한 검증
대상 경로에 쓰기 권한이 없으면 설치 실패. 사전에 권한 체크 후 안내.
주의
settings.local.json 가이드
설치 후 사용자가 settings.local.json을 직접 생성해야 함. 어떤 키(env.SLACK_WEBHOOK_URL 등)가 필요한지 명확히 안내.
주의
설치 토글 기본값
기본값은 OFF. 사용자가 의도적으로 켜야 설치 진행. 실수로 기존 .claude를 덮어쓰는 것 방지.
참고
파일 목록에서 제외 대상 시각화
뷰어에서 settings.local.json은 "배포 제외" 표시를 하여 사용자에게 명확히 인지.
참고
대시보드 카드에 .claude 상태
프로젝트 카드에 작은 아이콘으로 .claude 설치 여부를 표시하면 관리 편의성 증가. (P2)
